cmake_minimum_required(VERSION 3.16)
project(doce_atelie LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
add_executable(cupcake-server backend/main.cpp)
target_link_libraries(cupcake-server PRIVATE SQLite::SQLite3 Threads::Threads)
target_compile_options(cupcake-server PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
add_executable(domain-tests tests/domain_test.cpp)
target_link_libraries(domain-tests PRIVATE SQLite::SQLite3 Threads::Threads)
add_test(NAME domain COMMAND domain-tests ${CMAKE_SOURCE_DIR}/dist/products.json)
